4//! Types and associated logic to use while persisting
5//! data to the database.
6
7use std::collections::BTreeMap;
8
9use async_trait::async_trait;
10use futures::{FutureExt, StreamExt};
11use tokio_util::sync::CancellationToken;
12use tracing::info;
13
14use crate::{errors::IndexerError, types::IndexerResult};
15
16pub(crate) const CHECKPOINT_COMMIT_BATCH_SIZE: usize = 100;
17pub(crate) const UNPROCESSED_CHECKPOINT_SIZE_LIMIT: usize = 1000;
18
19/// Defines the logic of writing operations to the database.
20///
21/// The writing can refer to one or multiple tables in the database.
22#[async_trait]
23pub trait Writer<T: Send + Sync + 'static>: Send + Sync {
24    /// Returns the writer name.
25    fn name(&self) -> String;
26
27    /// Commits batch of transformed data to DB.
28    async fn persist(&self, batch: Vec<T>) -> IndexerResult<()>;
29
30    /// Reads high watermark of the table DB.
31    async fn get_watermark_hi(&self) -> IndexerResult<Option<u64>>;
32
33    /// Sets high watermark of the table DB, also update metrics.
34    async fn set_watermark_hi(&self, watermark_hi: u64) -> IndexerResult<()>;
35
36    /// Gets the current max checkpoint that can be committed by the writer.
37    ///
38    /// This is for writers that have a predefined lag compared to the latest
39    /// checkpoint in the network.
40    ///
41    /// One use-case is the objects snapshot handler, which waits for the lag
42    /// between snapshot and latest checkpoint to reach a certain threshold.
43    ///
44    /// # Note
45    /// By default, returns `u64::MAX`, which means no extra waiting is needed
46    /// before committing.
47    async fn get_max_committable_checkpoint(&self) -> IndexerResult<u64> {
48        Ok(u64::MAX)
49    }
50
51    /// Processes the received data and persists it into a storage.
52    ///
53    /// - The data are received form the ingestion worker in which stage is
54    ///   transformed into something which can be directly committed into the
55    ///   database.
56    /// - The data received by this function are not guaranteed to be in order.
57    ///   The purpose of this function is to order the data by checkpoint
58    ///   sequence number and to ensure data committed are in order and
59    ///   contiguous.
60    ///
61    /// In addition, the method updates the watermark of the table of the data
62    /// is persisted to.
63    async fn persist_sequentially(
64        &self,
65        cp_receiver: iota_metrics::metered_channel::Receiver<(u64, T)>,
66        cancel: CancellationToken,
67    ) -> IndexerResult<()> {
68        let checkpoint_commit_batch_size = std::env::var("CHECKPOINT_COMMIT_BATCH_SIZE")
69            .ok()
70            .and_then(|val| val.parse().ok())
71            .unwrap_or(CHECKPOINT_COMMIT_BATCH_SIZE);
72        let mut stream = iota_metrics::metered_channel::ReceiverStream::new(cp_receiver)
73            .ready_chunks(checkpoint_commit_batch_size);
74
75        let mut unprocessed = BTreeMap::new();
76        let mut tuple_batch = vec![];
77        let mut next_cp_to_process = self
78            .get_watermark_hi()
79            .await?
80            .map(|n| n.saturating_add(1))
81            .unwrap_or_default();
82
83        loop {
84            if cancel.is_cancelled() {
85                info!("transform and load task terminating gracefully");
86                return Ok(());
87            }
88
89            // Try to fetch new data tuple from the stream
90            if unprocessed.len() >= UNPROCESSED_CHECKPOINT_SIZE_LIMIT {
91                tracing::debug!(
92                    "Unprocessed checkpoint size reached limit {UNPROCESSED_CHECKPOINT_SIZE_LIMIT}, skip reading from stream..."
93                );
94            } else {
95                // Try to fetch new data tuple from the stream
96                match stream.next().now_or_never() {
97                    Some(Some(tuple_chunk)) => {
98                        if cancel.is_cancelled() {
99                            info!("transform and load task terminating gracefully");
100                            return Ok(());
101                        }
102                        for (cp_seq, data) in tuple_chunk {
103                            unprocessed.insert(cp_seq, (cp_seq, data));
104                        }
105                    }
106                    Some(None) => break, // Stream has ended
107                    None => {}           // No new data tuple available right now
108                }
109            }
110
111            // Process unprocessed checkpoints, even no new checkpoints from stream
112            let checkpoint_lag_limiter = self.get_max_committable_checkpoint().await?;
113            while next_cp_to_process <= checkpoint_lag_limiter {
114                if let Some(data_tuple) = unprocessed.remove(&next_cp_to_process) {
115                    tuple_batch.push(data_tuple);
116                    next_cp_to_process += 1;
117                } else {
118                    break;
119                }
120            }
121
122            if !tuple_batch.is_empty() && checkpoint_lag_limiter != 0 {
123                let tuple_batch = std::mem::take(&mut tuple_batch);
124                let (last_checkpoint_seq, _data) = tuple_batch.last().unwrap();
125                let last_checkpoint_seq = last_checkpoint_seq.to_owned();
126                let batch = tuple_batch
127                    .into_iter()
128                    .map(|(_cp_seq, data)| data)
129                    .collect();
130                self.persist(batch).await.map_err(|e| {
131                    IndexerError::PostgresWrite(format!(
132                        "failed to load transformed data into DB for handler {}: {e}",
133                        self.name()
134                    ))
135                })?;
136                self.set_watermark_hi(last_checkpoint_seq).await?;
137            }
138        }
139        Err(IndexerError::ChannelClosed(format!(
140            "checkpoint channel is closed unexpectedly for handler {}",
141            self.name()
142        )))
143    }
144}
