5use std::fmt::Display;
6
7use consensus_core::BlockAPI;
8use iota_types::{digests::ConsensusCommitDigest, messages_consensus::ConsensusTransaction};
9
10use crate::consensus_types::AuthorityIndex;
11
12/// A list of tuples of:
13/// (block origin authority index, all transactions contained in the block).
14/// For each transaction, returns deserialized transaction and its serialized
15/// size.
16type ConsensusOutputTransactions = Vec<(AuthorityIndex, Vec<(ConsensusTransaction, usize)>)>;
17
18pub(crate) trait ConsensusOutputAPI: Display {
19    fn reputation_score_sorted_desc(&self) -> Option<Vec<(AuthorityIndex, u64)>>;
20    fn leader_round(&self) -> u64;
21    fn leader_author_index(&self) -> AuthorityIndex;
22
23    /// Returns epoch UNIX timestamp in milliseconds
24    fn commit_timestamp_ms(&self) -> u64;
25
26    /// Returns a unique global index for each committed sub-dag.
27    fn commit_sub_dag_index(&self) -> u64;
28
29    /// Returns all transactions in the commit.
30    fn transactions(&self) -> ConsensusOutputTransactions;
31
32    /// Returns the digest of consensus output.
33    fn consensus_digest(&self) -> ConsensusCommitDigest;
34}
35macro_rules! impl_consensus_output_api {
36    (
37        // Type to implement for
38        type = $ty:path,
39        // Fully qualified commit digest type used in the size assertion
40        commit_digest = $commit_digest:path,
41        // How to iterate items that carry (round/author/txs)
42        iterate = |$self_ident:ident, $item_ident:ident| $iter:expr,
43        // How to read the round from an item (u64 or cast yourself later)
44        round = |$round_item:ident| $round_expr:expr,
45        // How to read the author index value (u32/usize; we cast to AuthorityIndex)
46        author = |$auth_item:ident| $author_expr:expr,
47        // How to get the `&[u8]` txs iterator source (something with `.iter()` over tx buffers)
48        txs = |$txs_item:ident| $txs_expr:expr
49    ) => {
50        impl ConsensusOutputAPI for $ty {
51            fn reputation_score_sorted_desc(&self) -> Option<Vec<(AuthorityIndex, u64)>> {
52                if !self.reputation_scores_desc.is_empty() {
53                    Some(
54                        self.reputation_scores_desc
55                            .iter()
56                            .map(|(id, score)| (id.value() as AuthorityIndex, *score))
57                            .collect(),
58                    )
59                } else {
60                    None
61                }
62            }
63
64            fn leader_round(&self) -> u64 {
65                self.leader.round as u64
66            }
67
68            fn leader_author_index(&self) -> AuthorityIndex {
69                self.leader.author.value() as AuthorityIndex
70            }
71
72            fn commit_timestamp_ms(&self) -> u64 {
73                self.timestamp_ms
74            }
75
76            fn commit_sub_dag_index(&self) -> u64 {
77                self.commit_ref.index.into()
78            }
79
80            fn transactions(&self) -> ConsensusOutputTransactions {
81                let $self_ident = self;
82                ($iter)
83                    .map(|$item_ident| {
84                        let round = { $round_expr } as u64;
85                        let author = { $author_expr } as AuthorityIndex;
86
87                        let transactions: Vec<_> = ({
88                                let $txs_item = $item_ident;
89                                $txs_expr
90                            })
91                            .iter()
92                            .flat_map(|tx| {
93                                let transaction = bcs::from_bytes::<ConsensusTransaction>(tx.data());
94                                match transaction {
95                                    Ok(transaction) => Some((transaction, tx.data().len())),
96                                    Err(err) => {
97                                        tracing::error!(
98                                            "Failed to deserialize sequenced consensus transaction \
99                                             (this should not happen) {err} from {author} at {round}"
100                                        );
101                                        None
102                                    }
103                                }
104                            })
105                            .collect();
106
107                        (author, transactions)
108                    })
109                    .collect()
110            }
111
112            fn consensus_digest(&self) -> ConsensusCommitDigest {
113                // Ensure wire layout matches.
114                static_assertions::assert_eq_size!(ConsensusCommitDigest, $commit_digest);
115                ConsensusCommitDigest::new(self.commit_ref.digest.into_inner())
116            }
117        }
118    };
119}

121// ===== Use the macro for the two concrete types =====
122
123// consensus_core::CommittedSubDag:
124// - iterate over `self.blocks`
125// - per-item accessors: round()/author().value()/transactions()
126impl_consensus_output_api! {
127    type = consensus_core::CommittedSubDag,
128    commit_digest = consensus_core::CommitDigest,
129    iterate = |self_, block| self_.blocks.iter(),
130    round   = |block| block.round(),
131    author  = |block| block.author().value(),
132    txs     = |block| block.transactions()
133}
134
135// starfish_core::CommittedSubDag:
136// - iterate over `self.transactions` (VerifiedTransactions)
137// - per-item accessors via block_ref(): .round / .author.value()
138// - txs via vt.transactions()
139impl_consensus_output_api! {
140    type = starfish_core::CommittedSubDag,
141    commit_digest = starfish_core::CommitDigest,
142    iterate = |self_, vt| self_.transactions.iter(),
143    round   = |vt| vt.block_ref().round,
144    author  = |vt| vt.block_ref().author.value(),
145    txs     = |vt| vt.transactions()
146}
