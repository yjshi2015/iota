5//! This module defines the network interface, and provides network
6//! implementations for the consensus protocol.
7//!
8//! Having an abstract network interface allows
9//! - simplifying the semantics of sending data and serving requests over the
10//!   network
11//! - hiding implementation specific types and semantics from the consensus
12//!   protocol
13//! - allowing easy swapping of network implementations, for better performance
14//!   or testing
15//!
16//! When modifying the client and server interfaces, the principle is to keep
17//! the interfaces low level, close to underlying implementations in semantics.
18//! For example, the client interface exposes sending messages to a specific
19//! peer, instead of broadcasting to all peers. Subscribing to a stream of
20//! blocks gets back the stream via response, instead of delivering the stream
21//! directly to the server. This keeps the logic agnostics to the underlying
22//! network outside of this module, so they can be reused easily across network
23//! implementations.
24

62/// Network client for communicating with peers.
63///
64/// NOTE: the timeout parameters help saving resources at client and potentially
65/// server. But it is up to the server implementation if the timeout is honored.
66/// - To bound server resources, server should implement own timeout for
67///   incoming requests.
68#[async_trait]
69pub(crate) trait NetworkClient: Send + Sync + Sized + 'static {
70    /// Subscribes to blocks from a peer after last_received round.
71    #[allow(dead_code)]
72    async fn subscribe_block_bundles(
73        &self,
74        peer: AuthorityIndex,
75        last_received: Round,
76        timeout: Duration,
77    ) -> ConsensusResult<BlockBundleStream>;
78
79    /// Fetches transactions for the given block references from a peer.
80    async fn fetch_transactions(
81        &self,
82        peer: AuthorityIndex,
83        block_refs: Vec<BlockRef>,
84        timeout: Duration,
85    ) -> ConsensusResult<Vec<Bytes>>;
86
87    // TODO: add a parameter for maximum total size of blocks returned.
88    /// Fetches serialized `SignedBlockHeader`s from a peer. It also might
89    /// return additional ancestor blocks of the requested blocks according
90    /// to the provided `highest_accepted_rounds`. The
91    /// `highest_accepted_rounds` length should be equal to the committee
92    /// size. If `highest_accepted_rounds` is empty then it will be simply
93    /// ignored.
94    async fn fetch_block_headers(
95        &self,
96        peer: AuthorityIndex,
97        block_refs: Vec<BlockRef>,
98        highest_accepted_rounds: Vec<Round>,
99        timeout: Duration,
100    ) -> ConsensusResult<Vec<Bytes>>;
101
102    /// Fetches serialized commits in the commit range from a peer.
103    /// Returns a tuple of both the serialized commits, and serialized blocks
104    /// that contain votes certifying the last commit.
105    async fn fetch_commits(
106        &self,
107        peer: AuthorityIndex,
108        commit_range: CommitRange,
109        timeout: Duration,
110    ) -> ConsensusResult<(Vec<Bytes>, Vec<Bytes>)>;
111
112    /// Fetches the latest block from `peer` for the requested `authorities`.
113    /// The latest blocks are returned in the serialised format of
114    /// `SignedBlocks`. The method can return multiple blocks per peer as
115    /// its possible to have equivocations.
116    async fn fetch_latest_block_headers(
117        &self,
118        peer: AuthorityIndex,
119        authorities: Vec<AuthorityIndex>,
120        timeout: Duration,
121    ) -> ConsensusResult<Vec<Bytes>>;
122}

124/// Network service for handling requests from peers.
125#[async_trait]
126pub(crate) trait NetworkService: Send + Sync + 'static {
127    /// Handles the block and headers sent from the peer via subscription
128    /// stream. Peer value can be trusted to be a valid authority index. But
129    /// serialized_block must be verified before its contents are trusted.
130    async fn handle_subscribed_block_bundle(
131        &self,
132        peer: AuthorityIndex,
133        serialized_block_bundle: SerializedBlockBundle,
134        encoder: &mut Box<dyn ShardEncoder + Send + Sync>,
135    ) -> ConsensusResult<()>;
136
137    /// Handles the subscription request from the peer.
138    /// A stream of newly proposed blocks with additional data (headers or
139    /// shards) is returned to the peer. The stream continues until the end
140    /// of epoch, peer unsubscribes, or a network error / crash occurs.
141    async fn handle_subscribe_block_bundles_request(
142        &self,
143        peer: AuthorityIndex,
144        last_received: Round,
145    ) -> ConsensusResult<BlockBundleStream>;
146
147    /// Handles the request to fetch block headers by references from the peer.
148    async fn handle_fetch_headers(
149        &self,
150        peer: AuthorityIndex,
151        block_refs: Vec<BlockRef>,
152        highest_accepted_rounds: Vec<Round>,
153    ) -> ConsensusResult<Vec<Bytes>>;
154
155    /// Handles the request to fetch commits by index range from the peer.
156    async fn handle_fetch_commits(
157        &self,
158        peer: AuthorityIndex,
159        commit_range: CommitRange,
160    ) -> ConsensusResult<(Vec<TrustedCommit>, Vec<VerifiedBlockHeader>)>;
161
162    /// Handles the request to fetch the latest block headers for the provided
163    /// `authorities`.
164    async fn handle_fetch_latest_block_headers(
165        &self,
166        peer: AuthorityIndex,
167        authorities: Vec<AuthorityIndex>,
168    ) -> ConsensusResult<Vec<Bytes>>;
169
170    /// Handles the request to get the latest received & accepted rounds of all
171    /// authorities.
172    async fn handle_get_latest_rounds(
173        &self,
174        peer: AuthorityIndex,
175    ) -> ConsensusResult<(Vec<Round>, Vec<Round>)>;
176
177    /// Handles the request to fetch transactions by references from the peer.
178    async fn handle_fetch_transactions(
179        &self,
180        peer: AuthorityIndex,
181        block_refs: Vec<BlockRef>,
182    ) -> ConsensusResult<Vec<Bytes>>;
183}

254pub(crate) struct SerializedBlockBundleParts {
255    pub(crate) serialized_block: Bytes,
256    pub(crate) serialized_headers: Vec<Bytes>,
257    pub(crate) serialized_shards: Vec<Bytes>,
258    pub(crate) useful_headers_authors_bitmask: [u64; 4],
259    pub(crate) useful_shards_authors_bitmask: [u64; 4],
260}
261
262fn authority_set_to_bitmask(authorities: &BTreeSet<AuthorityIndex>) -> [u64; 4] {
263    let mut bitmask = [0u64; 4];
264    for authority_index in authorities {
265        let index = authority_index.value();
266        let array_index = index / 64;
267        let bit_pos = index % 64;
268        bitmask[array_index] |= 1u64 << bit_pos;
269    }
270    bitmask
271}
272
273fn bitmask_to_authority_set(bitmask: [u64; 4]) -> BTreeSet<AuthorityIndex> {
274    let mut set = BTreeSet::new();
275    for (array_index, &bits) in bitmask.iter().enumerate() {
276        let mut bits = bits;
277        let base = array_index * 64;
278        while bits != 0 {
279            let bit = bits.trailing_zeros() as usize;
280            set.insert(AuthorityIndex::from((base + bit) as u8));
281            bits &= bits - 1;
282        }
283    }
284    set
285}
286
287impl SerializedBlockBundleParts {
288    pub(crate) fn useful_headers_authors(&self) -> BTreeSet<AuthorityIndex> {
289        bitmask_to_authority_set(self.useful_headers_authors_bitmask)
290    }
291    pub(crate) fn useful_shards_authors(&self) -> BTreeSet<AuthorityIndex> {
292        bitmask_to_authority_set(self.useful_shards_authors_bitmask)
293    }
294}

301impl TryFrom<VerifiedBlock> for SerializedBlockBundleParts {
302    type Error = ConsensusError;
303    fn try_from(verified_block: VerifiedBlock) -> ConsensusResult<Self> {
304        let (serialized_block_header, serialized_transactions) = verified_block.serialized();
305        let serialized_header_and_transactions = SerializedHeaderAndTransactions {
306            serialized_block_header: serialized_block_header.clone(),
307            serialized_transactions: serialized_transactions.clone(),
308        };
309        let bytes = bcs::to_bytes(&serialized_header_and_transactions)
310            .map_err(ConsensusError::SerializationFailure)?;
311        Ok(Self {
312            serialized_block: Bytes::from(bytes),
313            serialized_headers: vec![],
314            serialized_shards: vec![],
315            useful_headers_authors_bitmask: [0u64; 4],
316            useful_shards_authors_bitmask: [0u64; 4],
317        })
318    }
319}

321impl TryFrom<BlockBundle> for SerializedBlockBundleParts {
322    type Error = ConsensusError;
323    fn try_from(block_bundle: BlockBundle) -> ConsensusResult<Self> {
324        let (serialized_block_header, serialized_transactions) =
325            block_bundle.verified_block.serialized();
326        let serialized_header_and_transactions = SerializedHeaderAndTransactions {
327            serialized_block_header: serialized_block_header.clone(),
328            serialized_transactions: serialized_transactions.clone(),
329        };
330        let bytes = bcs::to_bytes(&serialized_header_and_transactions)
331            .map_err(ConsensusError::SerializationFailure)?;
332        let mut serialized_block_headers = vec![];
333        for block_header in block_bundle.verified_headers.iter() {
334            serialized_block_headers.push(block_header.serialized().clone());
335        }
336        Ok(Self {
337            serialized_block: Bytes::from(bytes),
338            serialized_headers: serialized_block_headers,
339            serialized_shards: block_bundle.serialized_shards,
340            useful_headers_authors_bitmask: authority_set_to_bitmask(
341                &block_bundle.useful_headers_authors,
342            ),
343            useful_shards_authors_bitmask: authority_set_to_bitmask(
344                &block_bundle.useful_shards_authors,
345            ),
346        })
347    }
348}

389#[cfg(test)]
390mod tests {
391    use rand::{seq::IteratorRandom, thread_rng};
392
393    use super::*;
394    use crate::TestBlockHeader;
395    #[test]
396    fn test_block_bundle_useful_authorities_set_bitmask_conversion() {
397        let block = VerifiedBlock::new_for_test(TestBlockHeader::new(0u32, 0u8).build());
398        // Generate a random sample of AuthorityIndex values (from 0..=255).
399        let mut rng = thread_rng();
400        let useful_authorities: BTreeSet<AuthorityIndex> = (0u8..=255)
401            .choose_multiple(&mut rng, 50) // pick 50 random distinct authorities
402            .into_iter()
403            .map(AuthorityIndex::from)
404            .collect();
405
406        let block_bundle = BlockBundle {
407            verified_block: block,
408            verified_headers: vec![],
409            serialized_shards: vec![],
410            useful_headers_authors: useful_authorities.clone(),
411            useful_shards_authors: useful_authorities.clone(),
412        };
413        let serialized_bundle = SerializedBlockBundle::try_from(block_bundle).unwrap();
414        let serialized_bundle_parts =
415            SerializedBlockBundleParts::try_from(serialized_bundle).unwrap();
416        let converted_useful_authorities = serialized_bundle_parts.useful_headers_authors();
417        assert_eq!(useful_authorities, converted_useful_authorities);
418    }
