38enum CoreThreadCommand {
39    /// Add blocks to be processed and accepted
40    AddBlocks(
41        Vec<VerifiedBlock>,
42        oneshot::Sender<(
43            BTreeSet<BlockRef>,
44            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
45        )>,
46    ),
47    /// Add block headers to be processed and accepted
48    AddBlockHeaders(
49        Vec<VerifiedBlockHeader>,
50        oneshot::Sender<(
51            BTreeSet<BlockRef>,
52            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
53        )>,
54    ),
55    /// Add committed sub dag blocks for processing and acceptance.
56    AddCertifiedCommits(
57        CertifiedCommits,
58        oneshot::Sender<(
59            BTreeSet<BlockRef>,
60            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
61        )>,
62    ),
63    /// Called when the min round has passed or the leader timeout occurred and
64    /// a block should be produced. When the command is called with `force =
65    /// true`, then the block will be created for `round` skipping
66    /// any checks (ex leader existence of previous round). More information can
67    /// be found on the `Core` component.
68    NewBlock(
69        Round,
70        oneshot::Sender<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>>,
71        ReasonToCreateBlock,
72    ),
73    /// Request missing blocks that need to be synced together with authorities
74    /// that have these blocks.
75    GetMissingBlocks(oneshot::Sender<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>>),
76    /// Add transactions to be processed and accepted
77    AddTransactions(
78        Vec<VerifiedTransactions>,
79        oneshot::Sender<()>,
80        TransactionSource,
81    ),
82    /// Add shards to the dag_state
83    AddShards(Vec<VerifiedOwnShard>, oneshot::Sender<()>),
84    /// Get missing transaction data that need to be synced
85    GetMissingTransactionData(oneshot::Sender<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>>),
86}

164pub(crate) struct CoreThreadHandle {
165    sender: Sender<CoreThreadCommand>,
166    join_handle: tokio::task::JoinHandle<()>,
167}
168
169impl CoreThreadHandle {
170    pub async fn stop(self) {
171        // drop the sender, that will force all the other weak senders to not able to
172        // upgrade.
173        drop(self.sender);
174        self.join_handle.await.ok();
175    }
176}

178struct CoreThread {
179    core: Core,
180    receiver: Receiver<CoreThreadCommand>,
181    rx_quorum_subscribers_exists: watch::Receiver<bool>,
182    rx_last_known_proposed_round: watch::Receiver<Round>,
183    context: Arc<Context>,
184}
185
186impl CoreThread {
187    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %self.context.own_index)))]
188    pub async fn run(mut self) -> ConsensusResult<()> {
189        tracing::debug!("Started core thread");
190
191        loop {
192            tokio::select! {
193                command = self.receiver.recv() => {
194                    let Some(command) = command else {
195                        break;
196                    };
197                    self.context.metrics.node_metrics.core_lock_dequeued.inc();
198                    match command {
199                        CoreThreadCommand::AddBlocks(blocks, sender) => {
200                            let _scope = monitored_scope("CoreThread::loop::add_blocks");
201                            let (missing_block_refs, missing_committed_txns) = self.core.add_blocks(blocks)?;
202                            sender.send((missing_block_refs, missing_committed_txns)).ok();
203                        }
204                        CoreThreadCommand::AddBlockHeaders(block_headers, sender) => {
205                            let _scope = monitored_scope("CoreThread::loop::add_block_headers");
206                            let (missing_block_refs, missing_committed_txns) = self.core.add_block_headers(block_headers)?;
207                            sender.send((missing_block_refs, missing_committed_txns)).ok();
208                        }
209                        CoreThreadCommand::AddCertifiedCommits(commits, sender) => {
210                            let _scope = monitored_scope("CoreThread::loop::add_certified_commits");
211                            let (missing_block_refs, missing_committed_txns) = self.core.add_certified_commits(commits)?;
212                            sender.send((missing_block_refs, missing_committed_txns)).ok();
213                        }
214                        CoreThreadCommand::NewBlock(round, sender, reason) => {
215                            let _scope = monitored_scope("CoreThread::loop::new_block");
216                            let (_new_block_opt, missing_committed_txns) = self.core.new_block(round, reason)?;
217                            sender.send(missing_committed_txns).ok();
218                        }
219                        CoreThreadCommand::GetMissingBlocks(sender) => {
220                            let _scope = monitored_scope("CoreThread::loop::get_missing_blocks");
221                            sender.send(self.core.get_missing_blocks()).ok();
222                        }
223                        CoreThreadCommand::AddTransactions(transactions, sender, source) => {
224                            let _scope = monitored_scope("CoreThread::loop::add_transactions");
225                            self.core.add_transactions(transactions, source)?;
226                            sender.send(()).ok();
227                        }
228                        CoreThreadCommand::AddShards(serialized_shards, sender) => {
229                            let _scope = monitored_scope("CoreThread::loop::add_shards");
230                            self.core.add_shards(serialized_shards)?;
231                            sender.send(()).ok();
232                        }
233                        CoreThreadCommand::GetMissingTransactionData(sender) => {
234                            let _scope = monitored_scope("CoreThread::loop::get_missing_transaction_data");
235                            sender.send(self.core.get_missing_transaction_data()).ok();
236                        }
237                    }
238                }
239                _ = self.rx_last_known_proposed_round.changed() => {
240                    let _scope = monitored_scope("CoreThread::loop::set_last_known_proposed_round");
241                    let round = *self.rx_last_known_proposed_round.borrow();
242                    self.core.set_last_known_proposed_round(round);
243                    self.core.new_block(round + 1, ReasonToCreateBlock::KnownLastBlock)?;
244                }
245                _ = self.rx_quorum_subscribers_exists.changed() => {
246                    let _scope = monitored_scope("CoreThread::loop::set_quorum_subscribers_exists");
247                    let should_propose_before = self.core.should_propose();
248                    let exists = *self.rx_quorum_subscribers_exists.borrow();
249                    self.core.set_quorum_subscribers_exists(exists);
250                    if !should_propose_before && self.core.should_propose() {
251                        // If core cannot propose before but can propose now, try to produce a new block to ensure liveness,
252                        // because block proposal could have been skipped.
253                        self.core.new_block(Round::MAX, ReasonToCreateBlock::QuorumSubscribersExist)?;
254                    }
255                }
256            }
257        }
258
259        Ok(())
260    }
261}

263#[derive(Clone)]
264pub(crate) struct ChannelCoreThreadDispatcher {
265    context: Arc<Context>,
266    sender: WeakSender<CoreThreadCommand>,
267    tx_quorum_subscribers_exists: Arc<watch::Sender<bool>>,
268    tx_last_known_proposed_round: Arc<watch::Sender<Round>>,
269    highest_received_rounds: Arc<Vec<AtomicU32>>,
270}
271
272impl ChannelCoreThreadDispatcher {
273    /// Starts the core thread for the consensus authority and returns a
274    /// dispatcher and handle for managing the core thread.
275    pub(crate) fn start(
276        context: Arc<Context>,
277        dag_state: &RwLock<DagState>,
278        core: Core,
279    ) -> (Self, CoreThreadHandle) {
280        // Initialize highest received rounds.
281        let highest_received_rounds = {
282            let dag_state = dag_state.read();
283            let highest_received_rounds = context
284                .committee
285                .authorities()
286                .map(|(index, _)| {
287                    AtomicU32::new(dag_state.get_last_block_header_for_authority(index).round())
288                })
289                .collect();
290
291            highest_received_rounds
292        };
293        let (sender, receiver) =
294            channel("consensus_core_commands", CORE_THREAD_COMMANDS_CHANNEL_SIZE);
295        let (tx_quorum_subscribers_exists, mut rx_quorum_subscribers_exists) =
296            watch::channel(false);
297        let (tx_last_known_proposed_round, mut rx_last_known_proposed_round) = watch::channel(0);
298        rx_quorum_subscribers_exists.mark_unchanged();
299        rx_last_known_proposed_round.mark_unchanged();
300        let core_thread = CoreThread {
301            core,
302            receiver,
303            rx_quorum_subscribers_exists,
304            rx_last_known_proposed_round,
305            context: context.clone(),
306        };
307
308        let join_handle = spawn_logged_monitored_task!(
309            async move {
310                if let Err(err) = core_thread.run().await {
311                    if !matches!(err, ConsensusError::Shutdown) {
312                        panic!("Fatal error occurred: {err}");
313                    }
314                }
315            },
316            "ConsensusCoreThread"
317        );
318
319        // Explicitly using downgraded sender in order to allow sharing the
320        // CoreThreadDispatcher but able to shutdown the CoreThread by dropping
321        // the original sender.
322        let dispatcher = ChannelCoreThreadDispatcher {
323            context,
324            sender: sender.downgrade(),
325            tx_quorum_subscribers_exists: Arc::new(tx_quorum_subscribers_exists),
326            tx_last_known_proposed_round: Arc::new(tx_last_known_proposed_round),
327            highest_received_rounds: Arc::new(highest_received_rounds),
328        };
329        let handle = CoreThreadHandle {
330            join_handle,
331            sender,
332        };
333        (dispatcher, handle)
334    }
335
336    async fn send(&self, command: CoreThreadCommand) {
337        self.context.metrics.node_metrics.core_lock_enqueued.inc();
338        if let Some(sender) = self.sender.upgrade() {
339            if let Err(err) = sender.send(command).await {
340                warn!(
341                    "Couldn't send command to core thread, probably is shutting down: {}",
342                    err
343                );
344            }
345        }
346    }
347}

349#[async_trait]
350impl CoreThreadDispatcher for ChannelCoreThreadDispatcher {
351    async fn add_blocks(
352        &self,
353        blocks: Vec<VerifiedBlock>,
354    ) -> Result<
355        (
356            BTreeSet<BlockRef>,
357            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
358        ),
359        CoreError,
360    > {
361        for block in &blocks {
362            self.highest_received_rounds[block.author()].fetch_max(block.round(), Ordering::AcqRel);
363        }
364        let (sender, receiver) = oneshot::channel();
365        self.send(CoreThreadCommand::AddBlocks(blocks, sender))
366            .await;
367        Ok(receiver.await.map_err(|e| Shutdown(e.to_string()))?)
368    }
369
370    async fn add_block_headers(
371        &self,
372        block_headers: Vec<VerifiedBlockHeader>,
373    ) -> Result<
374        (
375            BTreeSet<BlockRef>,
376            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
377        ),
378        CoreError,
379    > {
380        let (sender, receiver) = oneshot::channel();
381        self.send(CoreThreadCommand::AddBlockHeaders(block_headers, sender))
382            .await;
383        Ok(receiver.await.map_err(|e| Shutdown(e.to_string()))?)
384    }
385
386    async fn add_transactions(
387        &self,
388        transactions: Vec<VerifiedTransactions>,
389        source: TransactionSource,
390    ) -> Result<(), CoreError> {
391        let (sender, receiver) = oneshot::channel();
392        self.send(CoreThreadCommand::AddTransactions(
393            transactions,
394            sender,
395            source,
396        ))
397        .await;
398        receiver.await.map_err(|e| Shutdown(e.to_string()))
399    }
400
401    async fn add_shards(&self, shards: Vec<VerifiedOwnShard>) -> Result<(), CoreError> {
402        let (sender, receiver) = oneshot::channel();
403        self.send(CoreThreadCommand::AddShards(shards, sender))
404            .await;
405        receiver.await.map_err(|e| Shutdown(e.to_string()))
406    }
407
408    async fn get_missing_transaction_data(
409        &self,
410    ) -> Result<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>, CoreError> {
411        let (sender, receiver) = oneshot::channel();
412        self.send(CoreThreadCommand::GetMissingTransactionData(sender))
413            .await;
414        receiver.await.map_err(|e| Shutdown(e.to_string()))
415    }
416
417    async fn add_certified_commits(
418        &self,
419        commits: CertifiedCommits,
420    ) -> Result<
421        (
422            BTreeSet<BlockRef>,
423            BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
424        ),
425        CoreError,
426    > {
427        for commit in commits.commits() {
428            for block in commit.blocks() {
429                self.highest_received_rounds[block.author()]
430                    .fetch_max(block.round(), Ordering::AcqRel);
431            }
432        }
433        let (sender, receiver) = oneshot::channel();
434        self.send(CoreThreadCommand::AddCertifiedCommits(commits, sender))
435            .await;
436        Ok(receiver.await.map_err(|e| Shutdown(e.to_string()))?)
437    }
438
439    async fn new_block(
440        &self,
441        round: Round,
442        reason: ReasonToCreateBlock,
443    ) -> Result<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>, CoreError> {
444        let (sender, receiver) = oneshot::channel();
445        self.send(CoreThreadCommand::NewBlock(round, sender, reason))
446            .await;
447        receiver.await.map_err(|e| Shutdown(e.to_string()))
448    }
449
450    async fn get_missing_block_headers(
451        &self,
452    ) -> Result<BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>, CoreError> {
453        let (sender, receiver) = oneshot::channel();
454        self.send(CoreThreadCommand::GetMissingBlocks(sender)).await;
455        receiver.await.map_err(|e| Shutdown(e.to_string()))
456    }
457
458    fn set_quorum_subscribers_exists(&self, exists: bool) -> Result<(), CoreError> {
459        self.tx_quorum_subscribers_exists
460            .send(exists)
461            .map_err(|e| Shutdown(e.to_string()))
462    }
463
464    fn set_last_known_proposed_round(&self, round: Round) -> Result<(), CoreError> {
465        self.tx_last_known_proposed_round
466            .send(round)
467            .map_err(|e| Shutdown(e.to_string()))
468    }
469
470    fn highest_received_rounds(&self) -> Vec<Round> {
471        self.highest_received_rounds
472            .iter()
473            .map(|round| round.load(Ordering::Relaxed))
474            .collect()
475    }
476}
