15#[derive(Debug, Clone, PartialEq, Eq)]
16pub(crate) enum ExternalKeySource {
17    Ledger,
18    Unknown(String),
19}
20
21impl ExternalKeySource {
22    pub(crate) fn as_str(&self) -> &str {
23        match self {
24            ExternalKeySource::Ledger => "ledger",
25            ExternalKeySource::Unknown(source) => source.as_str(),
26        }
27    }
28}
29
30impl From<&str> for ExternalKeySource {
31    fn from(s: &str) -> Self {
32        match s {
33            "ledger" => ExternalKeySource::Ledger,
34            other => ExternalKeySource::Unknown(other.to_string()),
35        }
36    }
37}
38
39impl fmt::Display for ExternalKeySource {
40    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41        f.write_str(self.as_str())
42    }
43}

45pub(crate) async fn sign_transaction(
46    context: &mut WalletContext,
47    tx_data: &TransactionData,
48    signer_address: &IotaAddress,
49) -> Result<Signature> {
50    let iota_client = context.get_client().await?;
51
52    let key = context.config().keystore().get_key(signer_address)?;
53
54    match key {
55        StoredKey::KeyPair(_) => Ok(context.config().keystore().sign_secure(
56            signer_address,
57            tx_data,
58            Intent::iota_transaction(),
59        )?),
60        StoredKey::External {
61            derivation_path,
62            source,
63            ..
64        } => {
65            match ExternalKeySource::from(source.as_str()) {
66                ExternalKeySource::Ledger => {
67                    let Some(derivation_path) = derivation_path else {
68                        bail!(
69                            "Derivation path is required for Ledger signing. Please specify it in the keystore."
70                        );
71                    };
72
73                    let signer =
74                        LedgerSigner::new_with_default(derivation_path.clone(), Some(iota_client))?;
75                    // pass the transaction sender to the signer to ensure the correct
76                    // key is used
77                    Ok(signer
78                        .sign_transaction(tx_data, signer_address)
79                        .await
80                        .map(|s| s.signature)?)
81                }
82                ExternalKeySource::Unknown(name) => {
83                    bail!("External signing is not supported for source: {name}")
84                }
85            }
86        }
87    }
88}

90pub(crate) fn sign_secure<T>(
91    keystore: &impl AccountKeystore,
92    address: &IotaAddress,
93    msg: &T,
94    intent: Intent,
95) -> Result<Signature>
96where
97    T: Serialize,
98{
99    let key = keystore.get_key(address)?;
100    match key {
101        StoredKey::KeyPair(_) => Ok(keystore.sign_secure(address, &msg, intent)?),
102        StoredKey::External {
103            derivation_path,
104            source,
105            ..
106        } => {
107            match ExternalKeySource::from(source.as_str()) {
108                ExternalKeySource::Ledger => {
109                    let Some(derivation_path) = derivation_path else {
110                        bail!(
111                            "Derivation path is required for Ledger signing. Please specify it in the keystore."
112                        );
113                    };
114
115                    let ledger = Ledger::new_with_default()?;
116                    // Pass the expected address to the ledger to ensure the signature is for
117                    // the correct address.
118                    Ok(ledger
119                        .sign_intent(derivation_path, address, intent, &msg, vec![])?
120                        .signature)
121                }
122                ExternalKeySource::Unknown(name) => {
123                    bail!("External signing is not supported for source: {name}")
124                }
125            }
126        }
127    }
