4use futures::{Stream, StreamExt};
5use iota_grpc_types::{
6    checkpoints::{CertifiedCheckpointSummary, CheckpointData},
7    v0::checkpoints as grpc_checkpoints,
8};
9use tonic::transport::Channel;
10
11/// Enum representing the content of a checkpoint, either full data or summary.
12#[derive(Debug, Clone)]
13pub enum CheckpointContent {
14    Data(CheckpointData),
15    Summary(CertifiedCheckpointSummary),
16}
17
18/// Dedicated client for checkpoint-related gRPC operations.
19///
20/// This client handles all checkpoint service interactions including streaming
21/// checkpoints and querying epoch information.
22#[derive(Clone)]
23pub struct CheckpointClient {
24    client: grpc_checkpoints::checkpoint_service_client::CheckpointServiceClient<Channel>,
25}
26
27impl CheckpointClient {
28    /// Create a new CheckpointClient from a shared gRPC channel.
29    pub(super) fn new(channel: Channel) -> Self {
30        Self {
31            client: grpc_checkpoints::checkpoint_service_client::CheckpointServiceClient::new(
32                channel,
33            ),
34        }
35    }
36
37    /// Stream checkpoints with automatic deserialization.
38    ///
39    /// # Arguments
40    /// * `start_sequence_number` - Optional starting sequence number
41    /// * `end_sequence_number` - Optional ending sequence number
42    /// * `full` - Whether to stream full checkpoint data or just summaries
43    ///
44    /// # Returns
45    /// A stream of checkpoint content (either data or summaries)
46    pub async fn stream_checkpoints(
47        &mut self,
48        start_sequence_number: Option<u64>,
49        end_sequence_number: Option<u64>,
50        full: bool,
51    ) -> Result<impl Stream<Item = Result<CheckpointContent, tonic::Status>>, tonic::Status> {
52        let request = grpc_checkpoints::CheckpointStreamRequest {
53            start_sequence_number,
54            end_sequence_number,
55            full,
56        };
57        let stream = self.client.stream_checkpoints(request).await?.into_inner();
58
59        Ok(stream.map(|result| {
60            result.and_then(|checkpoint| {
61                Self::deserialize_checkpoint(&checkpoint).map_err(|e| {
62                    tonic::Status::internal(format!("Failed to deserialize checkpoint: {e}"))
63                })
64            })
65        }))
66    }
67
68    /// Get the first checkpoint sequence number for a given epoch.
69    pub async fn get_epoch_first_checkpoint_sequence_number(
70        &mut self,
71        epoch: u64,
72    ) -> Result<u64, tonic::Status> {
73        let request = grpc_checkpoints::EpochRequest { epoch };
74        let response = self
75            .client
76            .get_epoch_first_checkpoint_sequence_number(request)
77            .await?;
78        Ok(response.into_inner().sequence_number)
79    }
80
81    // ========================================
82    // Private Helper Methods
83    // ========================================
84
85    /// Deserialize checkpoint data based on the checkpoint type (full or
86    /// summary). Returns either checkpoint data or summary depending on the
87    /// checkpoint type.
88    fn deserialize_checkpoint(
89        checkpoint: &grpc_checkpoints::Checkpoint,
90    ) -> anyhow::Result<CheckpointContent> {
91        let bcs_data = checkpoint
92            .bcs_data
93            .as_ref()
94            .ok_or_else(|| anyhow::anyhow!("Missing BCS data in checkpoint"))?;
95
96        if checkpoint.is_full {
97            let checkpoint_data = bcs_data.deserialize_into::<CheckpointData>()?;
98            Ok(CheckpointContent::Data(checkpoint_data))
99        } else {
100            let checkpoint_summary = bcs_data.deserialize_into::<CertifiedCheckpointSummary>()?;
101            Ok(CheckpointContent::Summary(checkpoint_summary))
102        }
103    }
104}
