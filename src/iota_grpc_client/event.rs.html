4use anyhow::anyhow;
5use futures::{Stream, StreamExt};
6use iota_grpc_types::v0::events as grpc_events;
7use iota_json_rpc_types::{BcsEvent, IotaEvent};
8use iota_types::{
9    base_types::{IotaAddress, ObjectID, TransactionDigest},
10    event::EventID,
11};
12use move_core_types::{identifier::Identifier, language_storage::StructTag};
13use tonic::transport::Channel;
14
15/// Dedicated client for event-related gRPC operations.
16///
17/// This client handles all event service interactions including streaming
18/// events with filtering capabilities.
19#[derive(Clone)]
20pub struct EventClient {
21    client: grpc_events::event_service_client::EventServiceClient<Channel>,
22}
23
24impl EventClient {
25    /// Create a new EventClient from a shared gRPC channel.
26    pub(super) fn new(channel: Channel) -> Self {
27        Self {
28            client: grpc_events::event_service_client::EventServiceClient::new(channel),
29        }
30    }
31
32    /// Stream events with automatic BCS deserialization and filtering.
33    ///
34    /// # Arguments
35    /// * `filter` - Event filter to apply to the stream
36    ///
37    /// # Returns
38    /// A stream of IOTA events that match the specified filter
39    pub async fn stream_events(
40        &mut self,
41        filter: grpc_events::EventFilter,
42    ) -> Result<impl Stream<Item = Result<IotaEvent, tonic::Status>>, tonic::Status> {
43        let request = grpc_events::EventStreamRequest {
44            filter: Some(filter),
45        };
46        let stream = self.client.stream_events(request).await?.into_inner();
47
48        Ok(stream.map(|result| {
49            result.and_then(|event| {
50                Self::deserialize_event(&event).map_err(|e| {
51                    tonic::Status::internal(format!("Failed to deserialize event: {e}"))
52                })
53            })
54        }))
55    }
56
57    /// Deserialize event data from BCS bytes.
58    fn deserialize_event(event: &grpc_events::Event) -> anyhow::Result<IotaEvent> {
59        let event_id = event
60            .event_id
61            .as_ref()
62            .ok_or_else(|| anyhow!("Missing event ID"))?;
63
64        let tx_digest = event_id
65            .tx_digest
66            .as_ref()
67            .ok_or_else(|| anyhow!("Missing transaction digest"))?;
68
69        let package_id = event
70            .package_id
71            .as_ref()
72            .ok_or_else(|| anyhow!("Missing package ID"))?;
73
74        let sender = event
75            .sender
76            .as_ref()
77            .ok_or_else(|| anyhow!("Missing sender"))?;
78
79        let bcs_data = event
80            .event_data
81            .as_ref()
82            .ok_or_else(|| anyhow!("Missing event data"))?;
83
84        // Parse the StructTag from string
85        let type_tag: StructTag = event
86            .type_name
87            .parse()
88            .map_err(|e| anyhow!("Failed to parse type tag: {e}"))?;
89
90        // Parse the JSON
91        let parsed_json: serde_json::Value = serde_json::from_str(&event.parsed_json)
92            .map_err(|e| anyhow!("Failed to parse JSON: {e}"))?;
93
94        Ok(IotaEvent {
95            id: EventID {
96                tx_digest: TransactionDigest::new(
97                    tx_digest
98                        .digest
99                        .clone()
100                        .try_into()
101                        .map_err(|_| anyhow!("Invalid transaction digest length"))?,
102                ),
103                event_seq: event_id.event_seq,
104            },
105            package_id: ObjectID::from_bytes(&package_id.address)
106                .map_err(|e| anyhow!("Invalid package ID: {e}"))?,
107            transaction_module: Identifier::new(event.transaction_module.clone())
108                .map_err(|e| anyhow!("Invalid transaction module: {e}"))?,
109            sender: IotaAddress::from_bytes(&sender.address)
110                .map_err(|e| anyhow!("Invalid sender address: {e}"))?,
111            type_: type_tag,
112            parsed_json,
113            bcs: BcsEvent::new(bcs_data.data.clone()),
114            timestamp_ms: event.timestamp_ms,
115        })
116    }
117}
